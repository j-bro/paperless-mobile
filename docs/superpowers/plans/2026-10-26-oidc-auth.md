# OIDC Auth Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add native OIDC SSO login (system browser + PKCE + allauth headless) alongside existing password/API-key login.

**Architecture:** Capability detection after server-URL entry gates an SSO button; new `lib/features/login/oidc/` module handles provider discovery, AppAuth browser flow, and allauth `X-Session-Token` exchange to obtain the existing DRF `Token`, then reuses `AuthenticationCubit.addUser()` unchanged.

**Tech Stack:** Flutter/Dart, `flutter_appauth: ^7.0.0`, `app_links: ^6.0.0`, `dio: ^5.9.0`, `flutter_bloc: ^9.1.1`, `go_router: ^16.2.4`, `flutter_secure_storage` via existing `EncryptedLocalStore`.

## Global Constraints

- Minimum server: paperless-ngx with PR #11756 allauth headless (verify live, document exact version; older servers must show old UI unchanged).
- Public OIDC client only: PKCE S256, no client secret bundled in app.
- System browser only via `flutter_appauth` (Custom Tabs / ASWebAuthenticationSession); embedded `webview_flutter` is forbidden for IdP credentials.
- Custom scheme `paperlessmobile://oidc-callback` registered on Android + iOS; IdP must also whitelist `https://<server>/accounts/oidc/<provider>/login/callback/`.
- Downstream unchanged: final credential is still DRF `Authorization: Token <token>` in `SessionManagerImpl.updateSettings()`; `UserCredentials.token` format unchanged.
- Secret-required IdPs (Google/MS/GitHub with secret) are out of scope v1 — show documented fallback to manual API-key.
- No custom server endpoints (`/api/auth/external-login/*` rejected upstream — do not implement).
- Redact tokens/user-ids in logs via existing `redactUserId` / logger utils; never log `code_verifier`, `code`, tokens.

```dart
import 'package:dio/dio.dart';
import 'package:paperless_mobile/features/login/oidc/model/oidc_provider.dart';

class OidcApi {
  final Dio _dio;
  OidcApi(this._dio);

  Future<List<OidcProvider>> fetchProviders(String serverUrl) async {
    try {
      final res = await _dio.get(
        '$serverUrl/api/profile/social_account_providers/',
        options: Options(validateStatus: (s) => s != null && s < 500),
      );
      if (res.statusCode == 404) return [];
      final data = res.data;
      final list = data is List ? data : (data is Map ? (data['results'] ?? data['providers'] ?? []) : []);
      if (list is! List) return [];
      return list.whereType<Map>().map((e) => OidcProvider.fromJson(Map<String, dynamic>.from(e))).toList();
    } on DioException {
      return [];
    }
  }
}
```

```dart
// append to lib/api/modules/oidc_api.dart
class OidcSessionExpired implements Exception {}
class OidcSignupRequired implements Exception {
  final String? sessionToken;
  OidcSignupRequired(this.sessionToken);
}

class OidcExchangeResult {
  final String? sessionToken;
  final String? accessToken;
  final bool mfaRequired;
  const OidcExchangeResult({this.sessionToken, this.accessToken, this.mfaRequired = false});
}

extension OidcHeadlessApi on OidcApi {
  Map<String, String> _headers(String? sessionToken) =>
      sessionToken == null ? {} : {'X-Session-Token': sessionToken};

  OidcExchangeResult _parse(dynamic data) {
    final meta = (data is Map ? data['meta'] : null) as Map? ?? {};
    final flows = (data is Map ? (data['data'] as Map?)?['flows'] : null) as List? ?? [];
    final hasMfa = flows.any((f) => (f as Map)['id'] == 'mfa_authenticate');
    final hasSignup = flows.any((f) => (f as Map)['id'] == 'provider_signup');
    if (hasSignup) throw OidcSignupRequired(meta['session_token'] as String?);
    return OidcExchangeResult(
      sessionToken: meta['session_token'] as String?,
      accessToken: meta['access_token'] as String?,
      mfaRequired: hasMfa,
    );
  }

  Future<OidcExchangeResult> exchangeProviderToken({
    required String providerId,
    required String idToken,
    String? accessToken,
    String? sessionToken,
  }) async {
    final res = await _dio.post(
      '/_allauth/app/v1/auth/provider/token',
      data: {'provider': providerId, 'token': {'id_token': idToken, if (accessToken != null) 'access_token': accessToken}},
      options: Options(headers: _headers(sessionToken), validateStatus: (s) => s != null && s < 500),
    );
    if (res.statusCode == 410) throw OidcSessionExpired();
    return _parse(res.data);
  }

  Future<OidcExchangeResult> confirmMfa({required String code, required String sessionToken}) async {
    final res = await _dio.post(
      '/_allauth/app/v1/auth/mfa/authenticate',
      data: {'code': code},
      options: Options(headers: _headers(sessionToken), validateStatus: (s) => s != null && s < 500),
    );
    if (res.statusCode == 410) throw OidcSessionExpired();
    return _parse(res.data);
  }
}
```

```dart
// lib/features/login/oidc/cubit/oidc_cubit.dart
import 'package:bloc/bloc.dart';
import 'package:paperless_mobile/api/modules/oidc_api.dart';
import 'package:paperless_mobile/features/login/oidc/cubit/oidc_state.dart';
import 'package:paperless_mobile/features/login/oidc/model/oidc_provider.dart';
import 'package:paperless_mobile/features/login/oidc/service/oidc_auth_facade.dart';

export 'oidc_state.dart';

class OidcCubit extends Cubit<OidcState> {
  final OidcApi _api;
  final OidcAuthFacade _facade;
  static const redirectUrl = 'paperlessmobile://oidc-callback';
  String? _sessionToken;

  OidcCubit(this._api, this._facade) : super(const OidcInitial());

  Future<void> start(OidcProvider provider) async {
    emit(const OidcExchanging());
    try {
      final r = await _facade.authorize(provider: provider, redirectUrl: redirectUrl);
      if (r == null) { emit(const OidcCancelled()); return; }
      final ex = await _api.exchangeProviderToken(providerId: provider.id, idToken: r.idToken, accessToken: r.accessToken);
      if (ex.accessToken != null) { emit(OidcSuccess(ex.accessToken!)); return; }
      if (ex.mfaRequired && ex.sessionToken != null) {
        _sessionToken = ex.sessionToken;
        emit(OidcOtpRequired(ex.sessionToken!));
        return;
      }
      emit(const OidcError('OIDC exchange did not return a token.'));
    } on OidcSignupRequired {
      emit(const OidcError('Account is not linked. Log in via web UI once to link your SSO identity, then retry.'));
    } on OidcSessionExpired {
      emit(const OidcError('Session expired. Please retry.'));
    } catch (e) {
      emit(OidcError(e.toString()));
    }
  }

  Future<void> confirmMfa(String code) async {
    final s = _sessionToken;
    if (s == null) { emit(const OidcError('No pending session.')); return; }
    emit(const OidcExchanging());
    try {
      final ex = await _api.confirmMfa(code: code, sessionToken: s);
      if (ex.accessToken != null) { emit(OidcSuccess(ex.accessToken!)); return; }
      emit(const OidcError('Invalid code.'));
    } on OidcSessionExpired {
      emit(const OidcError('Session expired. Please retry.'));
    } catch (e) {
      emit(OidcError(e.toString()));
    }
  }
}
```

```dart
import 'package:flutter_appauth/flutter_appauth.dart';
import 'package:paperless_mobile/features/login/oidc/model/oidc_provider.dart';
import 'package:paperless_mobile/features/login/oidc/service/oidc_auth_facade.dart';

class AppAuthOidcFacade implements OidcAuthFacade {
  final FlutterAppAuth _appAuth;
  AppAuthOidcFacade([FlutterAppAuth? appAuth]) : _appAuth = appAuth ?? const FlutterAppAuth();

  @override
  Future<OidcBrowserResult?> authorize({required OidcProvider provider, required String redirectUrl}) async {
    final discoveryUrl = provider.discoveryUrl;
    final clientId = provider.clientId;
    if (discoveryUrl == null || clientId == null || discoveryUrl.isEmpty || clientId.isEmpty) {
      throw StateError('Provider ${provider.id} is missing discovery URL or client ID.');
    }
    final result = await _appAuth.authorizeAndExchangeCode(
      AuthorizationTokenRequest(
        clientId,
        redirectUrl,
        discoveryUrl: discoveryUrl,
        scopes: const ['openid', 'profile', 'email'],
      ),
    );
    if (result == null) return null;
    if (result.idToken == null) throw StateError('IdP did not return an id_token.');
    return OidcBrowserResult(idToken: result.idToken!, accessToken: result.accessToken);
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_bloc/flutter_bloc.dart';
import 'package:paperless_mobile/api/modules/oidc_api.dart';
import 'package:paperless_mobile/core/security/session_manager_impl.dart';

import 'package:paperless_mobile/features/login/server_connection/model/header_entry.dart';
import 'package:paperless_mobile/routing/routes/auth_route.dart';

class OidcLoginButton extends StatelessWidget {
  final String serverUrl;
  final List<OidcProvider> providers;
  final ClientCertificate? clientCertificate;
  final List<HeaderEntry>? additionalHeaders;
  final bool forceVisibleForTest;
  const OidcLoginButton({super.key, required this.serverUrl, required this.providers, this.clientCertificate, this.additionalHeaders, this.forceVisibleForTest = false});

  @override
  Widget build(BuildContext context) {
    if (providers.isEmpty && !forceVisibleForTest) return const SizedBox.shrink();
    return BlocProvider(
      create: (_) => OidcCubit(OidcApi(SessionManagerImpl().client), AppAuthOidcFacade()),
      child: BlocConsumer<OidcCubit, OidcState>(
        listener: (context, state) {
          if (state is OidcSuccess) {
            AddUserRoute(serverUrl: serverUrl, $extra: AddUserRouteExtra(token: state.drfToken, clientCertificate: clientCertificate, additionalHeaders: additionalHeaders)).replace(context);
          } else if (state is OidcError) {
            ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('${state.message}\nIf your IdP requires a client secret, use an API key instead.')));
          }
        },
        builder: (context, state) {
          final busy = state is OidcExchanging;
          return Column(children: [
            FilledButton.tonalIcon(
              onPressed: busy ? null : () => context.read<OidcCubit>().start(providers.first),
              icon: busy ? const SizedBox(width: 20, height: 20, child: CircularProgressIndicator(strokeWidth: 2)) : const Icon(Icons.key),
              label: const Text('Continue with SSO'),
            ),
            if (state is OidcOtpRequired)
              Padding(
                padding: const EdgeInsets.only(top: 8),
                child: TextField(
                  decoration: const InputDecoration(labelText: 'Authenticator code (SSO second factor)'),
                  keyboardType: TextInputType.number,
                  maxLength: 8,
                  onSubmitted: (v) => context.read<OidcCubit>().confirmMfa(v.trim()),
                ),
              ),
          ]);
        },
      ),
    );
  }
}
```
